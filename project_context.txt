# PyCav: High-Performance Passive Cavitation Imaging in PyTorch

![Python 3.10+](https://img.shields.io/badge/python-3.10+-blue.svg)
![PyTorch](https://img.shields.io/badge/PyTorch-GPU_Accelerated-ee4c2c.svg)
![License](https://img.shields.io/badge/license-MIT-green.svg)

**PyCav** is a modern, GPU-accelerated Python library dedicated to Passive Cavitation Imaging (PCI) and Passive Acoustic Mapping (PAM). By leveraging PyTorch's tensor operations and autograd engine capabilities (while strictly managing VRAM), PyCav transitions traditional, slow CPU-based acoustic mapping (e.g., MATLAB scripts) into highly parallelized, lightning-fast CUDA workflows.

## Key Features


* **Cavitation Simulation**: Built-in simulators for both **Broadband (BBag)** and **Stable/Subharmonic** cavitation signatures.
* **Robust Beamformers**:
  * **DAS** (Delay-and-Sum)
  * **RCB** (Robust Capon Beamformer / Minimum Variance)
  * **FB / PISA** (Functional Beamforming)
* **Fast processing**: GPU friendly architecture

---

## 📦 Installation

It is recommended to use a virtual environment (`venv`, `conda`, or `uv`).

```bash
git clone https://github.com/celest-la/PyCav.git
cd PyCav
pip install -e .
```
*(Dependencies will be installed automatically).*

---

##  Quickstart
There are 2 demos, for 3D et 2D reconstruction.
2D: `demo_reconstruction2D.py`
3D: `demo_reconstruction3D.py`

---

## 🏗️ Project Architecture

* `pycav.core`: Foundational geometries (`Grid`, `Probe`).
* `pycav.physics`: Acoustic wave propagation and steering vector computations.
* `pycav.simulation`: RF signal generation (`StableSimulator`, `BBagSimulator`).
* `pycav.solvers`: 
  * `preproc`: Cross-Spectral Matrix (`compute_csm`).
  * `beamformers`: State-of-the-art PCI solvers (`DAS`, `RCB`, `FB`, etc.).
* `pycav.utils`: Plotting tools, colormaps (dB mapping), and decorators.

import os
from pathlib import Path

# File extension types to include as readable code/text
ALLOWED_EXTENSIONS = {
    ".py", ".pyw", ".txt", ".md", ".json", ".yaml", 
    ".yml", ".toml", ".ini", ".cfg", ".env.example", ".sh", ".sql"
}

# Directories and files to automatically exclude
EXCLUDED_DIRS = {
    ".git", "__pycache__", ".venv", "venv", "env", 
    ".idea", ".vscode", "build", "dist", "node_modules", ".pytest_cache"
}

EXCLUDED_FILES = {
    "export_project.py",  # Exclude this script itself
    "project_context.txt", # Exclude the output file if placed in root
    ".DS_Store", "poetry.lock", "Pipfile.lock"
}

def is_excluded(path: Path, root_dir: Path) -> bool:
    """Check if a file or folder should be skipped."""
    rel_parts = path.relative_to(root_dir).parts
    if any(part in EXCLUDED_DIRS for part in rel_parts):
        return True
    if path.name in EXCLUDED_FILES:
        return True
    return False

def generate_tree(dir_path: Path, prefix: str = "") -> str:
    """Generates an ASCII representation of the file directory structure."""
    tree_str = ""
    contents = sorted(
        [p for p in dir_path.iterdir() if not is_excluded(p, dir_path.parent)],
        key=lambda x: (not x.is_dir(), x.name.lower())
    )

    for index, path in enumerate(contents):
        is_last = (index == len(contents) - 1)
        connector = "└── " if is_last else "├── "
        
        tree_str += f"{prefix}{connector}{path.name}\n"

        if path.is_dir():
            extension_prefix = "    " if is_last else "│   "
            tree_str += generate_tree(path, prefix + extension_prefix)

    return tree_str

def export_codebase(project_root: str, output_file: str):
    """Parses project tree and outputs file contents into a single text file."""
    root = Path(project_root).resolve()
    out_path = Path(output_file).resolve()

    with open(out_path, "w", encoding="utf-8") as out:
        # 1. Output Header & Architecture Tree
        out.write("=================================================================\n")
        out.write(f"PROJECT ARCHITECTURE: {root.name}\n")
        out.write("=================================================================\n\n")
        out.write(f"{root.name}/\n")
        out.write(generate_tree(root))
        out.write("\n\n")

        # 2. Output File Contents
        out.write("=================================================================\n")
        out.write("FILE CONTENTS\n")
        out.write("=================================================================\n\n")

        for path in sorted(root.rglob("*")):
            if path.is_file() and not is_excluded(path, root):
                if path.suffix in ALLOWED_EXTENSIONS or path.name in {"Dockerfile", "Makefile"}:
                    try:
                        relative_path = path.relative_to(root)
                        out.write(f"--- START FILE: {relative_path} ---\n")
                        with open(path, "r", encoding="utf-8", errors="replace") as f:
                            out.write(f.read())
                        out.write(f"\n--- END FILE: {relative_path} ---\n\n")
                    except Exception as e:
                        print(f"Skipping {path} due to error: {e}")

    print(f"Project successfully exported to: {out_path}")

if __name__ == "__main__":
    # Uses current working directory by default
    current_dir = os.getcwd()
    output_filename = "project_context.txt"
    
    export_codebase(current_dir, output_filename)

from __future__ import annotations
from typing import Union, Tuple, Optional

import torch
class Grid:
    def __init__(
        self, 
        x_coords: Union[torch.Tensor, list, float], 
        z_coords: Union[torch.Tensor, list, float], 
        y_coords: Optional[Union[torch.Tensor, list, float]] = None, 
        device="cpu") -> None:
        self.device = torch.device(device)
        self.x = torch.as_tensor(x_coords, dtype=torch.float32, device=self.device)
        self.z = torch.as_tensor(z_coords, dtype=torch.float32, device=self.device)
        
        if y_coords is not None:
            self.y = torch.as_tensor(y_coords, dtype=torch.float32, device=self.device)
            # 3D Meshgrid [Nx, Ny, Nz]
            grid_x, grid_y, grid_z = torch.meshgrid(self.x, self.y, self.z, indexing='ij')
            
            self.n_pixels = grid_x.numel()
            self.positions = torch.zeros((self.n_pixels, 3), device=self.device)
            self.positions[:, 0] = grid_x.flatten()
            self.positions[:, 1] = grid_y.flatten()
            self.positions[:, 2] = grid_z.flatten()
            
            self.shape = (len(self.x), len(self.y), len(self.z))
            self.is_3d = True
        else:
            self.y = torch.zeros(1, device=self.device)
            grid_x, grid_z = torch.meshgrid(self.x, self.z, indexing='ij')
            
            self.n_pixels = grid_x.numel()
            self.positions = torch.zeros((self.n_pixels, 3), device=self.device)
            self.positions[:, 0] = grid_x.flatten()
            self.positions[:, 1] = 0.0
            self.positions[:, 2] = grid_z.flatten()
            
            self.shape = (len(self.x), len(self.z))
            self.is_3d = False

    @classmethod
    def from_limits(cls, x_lim: Tuple[float, float], z_lim: Tuple[float, float], step: float, device="cpu") -> "Grid":
        """
        Crée une grille uniforme à partir de bornes.
        x_lim: (min, max) en mètres
        z_lim: (min, max) en mètres
        step: pas entre pixels en mètres
        """
        x_coords = torch.arange(x_lim[0], x_lim[1] + step, step)
        z_coords = torch.arange(z_lim[0], z_lim[1] + step, step)
        return cls(x_coords, z_coords, device=device)

    @classmethod
    def from_verasonics(cls, p_struct: dict[str, Any], device: Union[str, torch.device] ="cpu") ->"Grid":
        """
        Fonction à compléter et tester
        """
        return cls(x_coords, z_coords, device=device)
    
    @classmethod
    def from_limits_3d(
        cls, 
        x_lim: Tuple[float, float], 
        y_lim: Tuple[float, float], 
        z_lim: Tuple[float, float], 
        step: float, 
        device="cpu"
    ) -> "Grid":
        """
        Crée une grille 3D uniforme [Nx x Ny x Nz].
        """
        x_coords = torch.arange(x_lim[0], x_lim[1] + step, step)
        y_coords = torch.arange(y_lim[0], y_lim[1] + step, step)
        z_coords = torch.arange(z_lim[0], z_lim[1] + step, step)
        return cls(x_coords=x_coords, z_coords=z_coords, y_coords=y_coords, device=device)

    @validation_probe

    def from_linear(cls, n_el: int = 128, pitch: float = 0.3, fs: float = 22e6, device: Union[str, torch.device] ="cpu",fc: Optional[torch.Tensor] = None) -> "Probe":
        """
        Crée une sonde linéaire centrée sur x=0.
        n_el  : Nombre d'éléments (ex: 128)
        pitch : Espacement entre les centres des éléments en mètres (ex: 0.3e-3)

        # (Ex: pour 2 éléments à pitch 0.3, on aura -0.15 et +0.15)
        x = (torch.arange(n_el) - (n_el - 1) / 2) * pitch
        
        # On crée la matrice des positions [N_elements, 3] (X, Y, Z)
        positions = torch.zeros((n_el, 3), device=device)
        positions[:, 0] = x
        # Y et Z restent à 0 pour une sonde linéaire dans le plan XZ
        
        return cls(positions=positions, fs=fs, fc=fc, device=device)

    @classmethod
    def from_3d_coordinates(
        cls, 
        xp: Union[list, np.ndarray, torch.Tensor], 
        yp: Union[list, np.ndarray, torch.Tensor], 
        zp: Union[list, np.ndarray, torch.Tensor], 
        fs: float, 
        fc: Optional[torch.Tensor] = None, 
        device: Union[str, torch.device] = "cpu"
    ) -> "Probe":
        """
        Crée une sonde 3D à partir de vecteurs de coordonnées (xp, yp, zp).
        """
        device = torch.device(device)
        xp_t = torch.as_tensor(xp, dtype=torch.float32, device=device).flatten()
        yp_t = torch.as_tensor(yp, dtype=torch.float32, device=device).flatten()
        zp_t = torch.as_tensor(zp, dtype=torch.float32, device=device).flatten()
        
        if not (len(xp_t) == len(yp_t) == len(zp_t)):
            raise ValueError("xp, yp, zp doivent avoir la même longueur.")
            
        positions = torch.stack([xp_t, yp_t, zp_t], dim=1)  # [Ne, 3]
        
        if fc is None:
            fc = torch.tensor([1.5e6, 5.0e6], dtype=torch.float32, device=device)
        elif not torch.is_tensor(fc):
            fc = torch.as_tensor(fc, dtype=torch.float32, device=device)
            
        return cls(positions=positions, fs=fs, fc=fc, device=device)

    @classmethod
    def from_param_dict(cls, param: dict[str, Any], device: Union[str, torch.device] = "cpu") -> "Probe":
        """
        Crée une sonde directement depuis le dictionnaire MATLAB `param`.
        """
        return cls.from_3d_coordinates(
            xp=param["xp"],
            yp=param["yp"],
            zp=param["zp"],
            fs=param["fs"],
            fc=param.get("fc", [1.5e6, 5.0e6]),
            device=device
        )

    @classmethod
    def from_matlab_file(cls, filepath: Union[str, Path], struct_name: str = "param", device: Union[str, torch.device] = "cpu") -> "Probe":
        """
        Loads a .mat file directly and constructs the Probe.
        """
        import scipy.io as sio
        mat_data = sio.loadmat(filepath, squeeze_me=True)
        raw = mat_data[struct_name]
        
        param = {}
        for name in raw.dtype.names:
            val = raw[name]
            param[name] = val.item() if getattr(val, 'shape', None) == () else val
            
        return cls.from_param_dict(param, device=device)

from __future__ import annotations
import torch
from typing import Union, List
from pycav.core.probe import Probe

@torch.no_grad()
def compute_steering_vectors(
    probe: Probe, 
    grid_positions: torch.Tensor, 
    freqs: Union[torch.Tensor, List[Union[float, torch.Tensor]], float], 
    c0: float = 1540.0
) -> torch.Tensor:
    
    # 1. Calcul de la matrice de distance [Pixels x Eléments]
    dist = torch.cdist(grid_positions, probe.positions, p=2)
    
    # 2. Calcul des délais [Pixels x Eléments]
    tau = dist / c0
    
    # 3. SÉCURITÉ : Nettoyage de `freqs`
    # Si l'utilisateur passe une liste (ex: [axf]), on extrait les valeurs en float purs
    if isinstance(freqs, list):
        freqs_clean = [f.item() if torch.is_tensor(f) else float(f) for f in freqs]
        f = torch.tensor(freqs_clean, dtype=torch.float32, device=dist.device)
    else:
        f = torch.as_tensor(freqs, dtype=torch.float32, device=dist.device)

    # 4. Passage au domaine fréquentiel [Fréquences x Pixels x Eléments]
    f = f.view(-1, 1, 1)
    
    # steer_vec = exp(-2j * pi * f * tau)
    # Déjà en complex64 car f et tau sont en float32 !
    A = torch.exp(-2j * torch.pi * f * tau.unsqueeze(0))
    
    return A

@torch.no_grad()
def compute_delay(
    probe: Probe, 
    grid_positions: torch.Tensor, 
    c0: float = 1540.0
) -> torch.Tensor:
    
    dist = torch.cdist(grid_positions, probe.positions, p=2)
    return dist / c0 * probe.fs

class getPositions:
    @classmethod
    @torch.no_grad()
    def random_ellipse(
        cls,
        x_position_lim: Union[float, Tuple[float, float], list, torch.Tensor] = 0e-3,
        z_position_lim: Union[float, Tuple[float, float], list, torch.Tensor] = 20e-3,
        x_size_lim: Union[float, Tuple[float, float], list, torch.Tensor] = 1e-3,
        z_size_lim: Union[float, Tuple[float, float], list, torch.Tensor] = 1e-3,
        angle: Optional[float] = 0,
        density: int = 100,
        device: Union[str, torch.device] = "cpu"
    ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor, torch.Tensor, Union[float, torch.Tensor]]:
        """Génère un nuage 2D de positions de bulles dans le plan X-Z (y=0)."""
        device = torch.device(device)

        def get_val(lim):
            lim_t = torch.as_tensor(lim, device=device, dtype=torch.float32)
            if lim_t.ndim == 0 or lim_t.shape[0] == 1:
                return lim_t.squeeze()
            return lim_t[0] + torch.rand(1, device=device) * (lim_t[1] - lim_t[0])

        x_size = get_val(x_size_lim)
        z_size = get_val(z_size_lim)

        z_position_lim_t = torch.as_tensor(z_position_lim, device=device, dtype=torch.float32)
        if z_position_lim_t.numel() == 1:
            z_center = z_position_lim_t
        elif z_position_lim_t.numel() == 2:
            z_center = z_position_lim_t[0] + z_size / 2 + torch.rand(1, device=device) * (
                z_position_lim_t[1] - z_position_lim_t[0] - z_size
            )
        else:
            raise ValueError("z_position_lim should be of size 1 or 2")

        x_position_lim_t = torch.as_tensor(x_position_lim, device=device, dtype=torch.float32)
        if x_position_lim_t.numel() == 1:
            x_center = x_position_lim_t
        elif x_position_lim_t.numel() == 2:
            x_center = x_position_lim_t[0] + x_size / 2 + torch.rand(1, device=device) * (
                x_position_lim_t[1] - x_position_lim_t[0] - x_size
            )
        else:
            raise ValueError("x_position_lim should be of size 1 or 2")

        center = torch.tensor([x_center, z_center], dtype=torch.float32, device=device)
        ellipse_axes = torch.tensor([x_size, z_size], dtype=torch.float32, device=device)

        number_bubble = int(torch.round(x_size * z_size * density * 1e6).item())

        pos_temp = -0.5 * ellipse_axes + ellipse_axes * torch.rand((number_bubble, 2), device=device)

        ind_in = (pos_temp[:, 0]**2 / (0.5 * x_size)**2 + pos_temp[:, 1]**2 / (0.5 * z_size)**2) <= 1
        pos_temp2 = pos_temp[ind_in, :]

        if angle is None:
            angle = 360 * torch.rand(1, device=device)
        angle_rad = torch.as_tensor(angle * torch.pi / 180, device=device)
        cos_a = torch.cos(angle_rad)
        sin_a = torch.sin(angle_rad)

        R = torch.tensor([[cos_a, -sin_a], [sin_a, cos_a]], dtype=torch.float32, device=device)
        bubbles_pos = torch.matmul(pos_temp2, R.t()) + center
        y = torch.zeros((bubbles_pos.shape[0], 1), device=device)
        bubbles_pos = torch.cat([bubbles_pos, y], dim=1)
        bubbles_pos = bubbles_pos[:, [0, 2, 1]]  # [X, Y, Z]

        return bubbles_pos, center, x_size, z_size, angle

    @classmethod
    @torch.no_grad()
    def random_ellipsoid(
        cls,
        x_position_lim: Union[float, Tuple[float, float]] = 0e-3,
        y_position_lim: Union[float, Tuple[float, float]] = 0e-3,
        z_position_lim: Union[float, Tuple[float, float]] = 20e-3,
        x_size_lim: Union[float, Tuple[float, float]] = 1e-3,
        y_size_lim: Union[float, Tuple[float, float]] = 1e-3,
        z_size_lim: Union[float, Tuple[float, float]] = 1e-3,
        density: int = 100,
        num_bubbles: Optional[int] = None,
        device: Union[str, torch.device] = "cpu"
    ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:
        """
        Génère un nuage 3D de positions de bulles distribuées uniformément dans un ellipsoïde [X, Y, Z].
        density est exprimée en bulles / mm^3.
        """
        device = torch.device(device)

        def get_val(lim):
            lim_t = torch.as_tensor(lim, device=device, dtype=torch.float32)
            if lim_t.numel() == 1:
                return lim_t.squeeze()
            return lim_t[0] + torch.rand(1, device=device) * (lim_t[1] - lim_t[0])

        x_size = get_val(x_size_lim)
        y_size = get_val(y_size_lim)
        z_size = get_val(z_size_lim)

        x_center = get_val(x_position_lim)
        y_center = get_val(y_position_lim)
        z_center = get_val(z_position_lim)

        center = torch.stack([x_center, y_center, z_center])
        axes = torch.stack([x_size, y_size, z_size])

        if num_bubbles is None:
            # Volume d'un ellipsoïde 3D = (4/3) * pi * a * b * c (en m^3)
            volume_m3 = (4.0 / 3.0) * torch.pi * (0.5 * x_size) * (0.5 * y_size) * (0.5 * z_size)
            # Conversion m^3 en mm^3 (facteur 1e9)
            number_bubble = int(torch.round(volume_m3 * density * 1e9).item())
            number_bubble = max(number_bubble, 1)
        else:
            number_bubble = int(num_bubbles)

        # Rejection sampling inside 3D ellipsoid
        pos_temp = -0.5 * axes + axes * torch.rand((number_bubble * 4, 3), device=device)
        norm_sq = (pos_temp[:, 0] / (0.5 * x_size))**2 + (pos_temp[:, 1] / (0.5 * y_size))**2 + (pos_temp[:, 2] / (0.5 * z_size))**2
        valid_pos = pos_temp[norm_sq <= 1.0][:number_bubble]

        bubbles_pos = valid_pos + center
        return bubbles_pos, center, axes

class Stable(BubbleSource):
    def __init__(self, fs: float, f_hifu: float, device: Union[str, torch.device] = "cpu") -> None:
        self.fs = fs
        self.f_hifu = f_hifu
        self.device = torch.device(device)

    @torch.no_grad()
    def get_kernel(self, alpha: float = 1.0, beta: float = 0.8, delta: float = 0.1) -> torch.Tensor:
        T = 1 / self.f_hifu
        t = torch.arange(0, 2 * T, 1/self.fs, device=self.device, dtype=torch.float32)

        if t.shape[0] % 2 == 0:
            t = torch.arange(0, 2 * T + 1/self.fs, 1/self.fs, device=self.device, dtype=torch.float32)

        s = torch.sin(2 * torch.pi * self.f_hifu * t)
        kernel = torch.where(s > 0, s * alpha, s * beta)
        
        mask_cycle2 = (t >= T)
        kernel[mask_cycle2] *= (1 - delta)
        
        window = torch.hann_window(len(t), periodic=False, device=self.device, dtype=torch.float32)
        return (kernel * window) / torch.norm(kernel * window)


class StableSimulator:
    def __init__(self, probe: Probe, bubble_positions: torch.Tensor, c0: float = 1540, fhifu: float = 1e6, device: Union[str, torch.device] = "cpu") -> None:
        self.device = torch.device(device)
        self.probe = probe
        self.bubble_positions = bubble_positions.to(self.device)  # (Nb, 3)
        self.c0 = c0
        self.fhifu = fhifu
        self.nb = bubble_positions.shape[0]

    @torch.no_grad()
    def _compute_batch_delays(self, batch_pos: torch.Tensor, t0: float = 0) -> torch.Tensor:
        dist = torch.cdist(self.probe.positions.to(self.device), batch_pos, p=2)
        return (dist / self.c0) * self.probe.fs + t0

    @torch.no_grad()
    def _get_source_events(self, n_cycles: int, batch_nb: int, A0: float, sd: float, subharmonic: bool = True) -> Tuple[torch.Tensor, torch.Tensor]:
        step = 2 if subharmonic else 1
        T_samples = self.probe.fs / self.fhifu
        t_emit = (torch.arange(0, n_cycles, step, device=self.device, dtype=torch.float32).float() + 0.5) * T_samples
        n_pulses = t_emit.shape[0]
        amps = A0 + sd * torch.randn(n_pulses, batch_nb, device=self.device, dtype=torch.float32)
        return t_emit, amps

    @torch.no_grad()
    def _splat_linear(self, rf_grid: torch.Tensor, t_arrival: torch.Tensor, amplitudes: torch.Tensor) -> None:
        n_samples, nelem = rf_grid.shape
        t_flat = t_arrival.reshape(-1)
        a_flat = amplitudes.reshape(-1)
        
        elem_idx = torch.arange(nelem, device=self.device).view(-1, 1, 1).expand_as(t_arrival).reshape(-1)
        t_floor = torch.floor(t_flat).long()
        frac = t_flat - t_floor.float()

        for offset, weight in [(0, 1 - frac), (1, frac)]:
            idx = t_floor + offset
            mask = (idx >= 0) & (idx < n_samples)
            rf_grid.index_put_(
                (idx[mask], elem_idx[mask]), 
                (a_flat * weight)[mask], 
                accumulate=True
            )

    @torch.no_grad()
    def get_delayed_dirac(self, n_cycles: int, A0: float = 1, t0: float = 0, t_acq: float = 100e-6, sd: float = 0.1, batch_size: int = 1000) -> torch.Tensor:
        nelem = self.probe.positions.shape[0]
        n_samples = int(t_acq * self.probe.fs)
        dirac_RF = torch.zeros(n_samples, nelem, device=self.device)

        for i in range(0, self.nb, batch_size):
            batch_pos = self.bubble_positions[i : i + batch_size]
            curr_batch_nb = batch_pos.shape[0]
            
            index_tau = self._compute_batch_delays(batch_pos, t0)
            index_dirac, batch_amps = self._get_source_events(n_cycles, curr_batch_nb, A0, sd)
            
            tau_final = (index_tau.unsqueeze(-1) + index_dirac.view(1, 1, -1)).permute(0, 2, 1)
            full_amps = batch_amps.unsqueeze(0).expand_as(tau_final)
            
            self._splat_linear(dirac_RF, tau_final, full_amps)
            
        return dirac_RF

    @torch.no_grad()
    def apply_kernel(self, dirac_rf: torch.Tensor, kernel: torch.Tensor) -> torch.Tensor:
        x = dirac_rf.t().unsqueeze(0)
        nelem = dirac_rf.shape[1]
        weight = kernel.view(1, 1, -1).repeat(nelem, 1, 1)
        rf_conv = torch.nn.functional.conv1d(x, weight, padding='same', groups=nelem)
        return rf_conv.squeeze(0).t()

    def apply_probe_filter(self, rf_tensor: torch.Tensor) -> torch.Tensor:
        n = rf_tensor.shape[0]
        fft_rf = torch.fft.rfft(rf_tensor, dim=0)
        freqs = torch.fft.rfftfreq(n, d=1/self.probe.fs, device=self.device)
        
        band = self.probe.fc.to(self.device)
        mask = (freqs >= band[0]) & (freqs <= band[1])
        fft_rf[~mask, :] = 0
        
        return torch.fft.irfft(fft_rf, n=n, dim=0)

    @torch.no_grad()
    def simulate(
        self, n_cycles: int = 200, t_acq: float = 100e-6, A0: float = 1, sd: float = 0.1, 
        alpha: float = 1.0, beta: float = 0.5, delta: float = 0.2, 
        noise_level: float = 0.001, apply_filter: bool = True, batch_size: int = 1000
    ) -> torch.Tensor:
        diracs = self.get_delayed_dirac(n_cycles=n_cycles, A0=A0, t_acq=t_acq, sd=sd, batch_size=batch_size)
        k_sub = Stable(self.probe.fs, self.fhifu, device=self.device).get_kernel(alpha=alpha, beta=beta, delta=delta)
        rf = self.apply_kernel(diracs, k_sub)
        
        if noise_level > 0:
            noise = noise_level * torch.randn_like(rf) * rf.max()
            rf = rf + noise
            
        if apply_filter:
            rf = self.apply_probe_filter(rf)
            
        return rf


class BBagSimulator:
    def __init__(self, probe: Probe, bubble_positions: torch.Tensor, c0: float = 1540, fhifu: float = 1e6, device: Union[str, torch.device] = "cpu") -> None:
        self.device = torch.device(device)
        self.probe = probe
        self.bubble_positions = bubble_positions.to(self.device)  # (Nb, 3)
        self.c0 = c0
        self.fhifu = fhifu
        self.nb = bubble_positions.shape[0]

    @torch.no_grad()
    def get_RF(self, n_cycles: int = 200, A0: float = 1, t0: float = 0, t_acq: float = 100e-6, batch_size: int = 1000) -> torch.Tensor:
        n_bin = int(t_acq * self.probe.fs)
        n_source_samples = int(n_cycles * self.probe.fs / self.fhifu)
        freq = torch.fft.rfftfreq(n_bin, d=1/self.probe.fs, device=self.device)
        
        delayed_fft = torch.zeros((len(freq), self.probe.n_elements), dtype=torch.complex64, device=self.device)
        
        for i in range(0, self.nb, batch_size):
            batch_pos = self.bubble_positions[i : i + batch_size]
            curr_batch_nb = batch_pos.shape[0]
            
            source_signals = A0 * torch.rand((n_source_samples, curr_batch_nb), device=self.device)
            source_fft = torch.fft.rfft(source_signals, dim=0, n=n_bin)
            
            steer_vec = compute_steering_vectors(probe=self.probe, grid_positions=batch_pos, freqs=freq, c0=self.c0)
            delayed_fft += torch.einsum('fb, fbe -> fe', source_fft, steer_vec)
            
        return torch.fft.irfft(delayed_fft, n=n_bin, dim=0)

    def apply_probe_filter(self, rf_tensor: torch.Tensor) -> torch.Tensor:
        n = rf_tensor.shape[0]
        fft_rf = torch.fft.rfft(rf_tensor, dim=0)
        freqs = torch.fft.rfftfreq(n, d=1/self.probe.fs, device=self.device)
        
        band = self.probe.fc.to(self.device)
        mask = (freqs >= band[0]) & (freqs <= band[1])
        fft_rf[~mask, :] = 0
        
        return torch.fft.irfft(fft_rf, n=n, dim=0)

    @torch.no_grad()
    def simulate(
        self, t_acq: float = 100e-6, n_cycles: int = 200, A0: float = 1, 
        batch_size: int = 1000, noise_level: float = 0.001, apply_filter: bool = True
    ) -> torch.Tensor:
        rf = self.get_RF(n_cycles=n_cycles, A0=A0, t_acq=t_acq, batch_size=batch_size)
        
        if noise_level > 0:
            noise = noise_level * torch.randn_like(rf) * rf.max()
            rf = rf + noise
            
        if apply_filter:
            rf = self.apply_probe_filter(rf)
            
        return rf

class BaseFrequencySolver:
    def __init__(self, probe: Probe, grid: Grid, c: float = 1540) -> None:
        self.probe = probe
        self.grid = grid
        self.c = c

    @torch.no_grad()
    def get_steering_vectors(self, freqs: Union[torch.Tensor, List[float], float]) -> torch.Tensor:
        return compute_steering_vectors(self.probe, self.grid.positions, freqs, self.c)

    def _format_output(self, img: torch.Tensor, average: bool) -> torch.Tensor:
        F = img.shape[0]
        img_reshaped = img.view(F, *self.grid.shape)
        if average:
            return img_reshaped.mean(dim=0)
        return img_reshaped


class DAS(BaseFrequencySolver):
    def solve(self, csm: torch.Tensor, freqs: Union[torch.Tensor, List[float], float], average: bool = True) -> torch.Tensor:
        a = self.get_steering_vectors(freqs)
        a = a / torch.tensor(self.probe.positions.shape[0])
        img = torch.einsum('fpi,fij,fpj->fp', a.conj(), csm, a).real
        return self._format_output(img, average)


class RCB(BaseFrequencySolver):
    @torch.no_grad()
    def solve(self, csm: torch.Tensor, freqs: Union[torch.Tensor, List[float], float], epsilon: float = 0.1, average: bool = True) -> torch.Tensor:
        a_all = self.get_steering_vectors(freqs)
        F, P, M = a_all.shape
        device = csm.device
        
        L_val, U = torch.linalg.eigh(csm)
        L_val = torch.clamp(L_val, min=1e-12)
        
        final_map = torch.zeros((F, P), device=device, dtype=torch.float32)
        eps_t = torch.tensor(epsilon, device=device, dtype=torch.float32)
        sqrt_eps = torch.sqrt(eps_t)

        for f in range(F):
            a = a_all[f]
            L = L_val[f].unsqueeze(0)
            U_f = U[f]
            
            sumU = a.conj() @ U_f
            abs_sumU2 = torch.abs(sumU)**2
            norm_a = torch.sqrt(torch.tensor(float(M), device=device, dtype=torch.float32)) 
            
            lb = (norm_a - sqrt_eps) / (L.max() * sqrt_eps)
            lamb_init = 0.5 * max(lb.item(), 1e-10)
            lamb = torch.full((P,), lamb_init, device=device, dtype=torch.float32)

            for _ in range(20):
                denom = 1 + L * lamb.unsqueeze(1)   
                f_val = torch.sum(abs_sumU2 / (denom**2), dim=1) - eps_t
                fp_val = -2 * torch.sum((L * abs_sumU2) / (denom**3), dim=1)
                
                lamb = lamb - f_val / fp_val
                lamb = lamb.clamp(min=1e-10)

            term_pow = (lamb.unsqueeze(1) / (1 + lamb.unsqueeze(1) * L))**2
            denom_pow = torch.sum(L * abs_sumU2 * term_pow, dim=1)
            map_t = 1.0 / torch.clamp(denom_pow, min=1e-12)

            weighting = 1.0 / (1 + L * lamb.unsqueeze(1))
            norm_ac2 = torch.sum(abs_sumU2 * (1 - weighting)**2, dim=1)
            final_map[f] = map_t * (norm_ac2 / M)

        return self._format_output(final_map, average)


class FB(BaseFrequencySolver):
    @torch.no_grad()
    def solve(self, csm: torch.Tensor, freqs: Union[torch.Tensor, List[float], float], r: float = 1, method: str = 'standard', average: bool = True) -> torch.Tensor:
        a = self.get_steering_vectors(freqs)
        norm = torch.linalg.vector_norm(a, dim=-1, keepdim=True)
        a = a / norm
        L, V = torch.linalg.eigh(csm)
        L = torch.clamp(L, min=1e-12)
        proj = torch.einsum('fpm,fmi->fpi', a.conj(), V)
        if r != 0:
            L_pow = L**(1/r)
            quad_form = torch.einsum('fpi,fi,fpi->fp', proj, L_pow, proj.conj()).real
            img = (quad_form**r) / self.probe.positions.shape[0]
        else:
            L_log = torch.log(L)
            W = torch.einsum('fmi,fi,fni->fmn', V, L_log, V.conj())
            log_form = torch.einsum('fpi,fij,fpj->fp', a.conj(), W, a).real
            img = torch.exp(log_form) / self.probe.positions.shape[0]           

        return self._format_output(img, average)

from __future__ import annotations

@torch.no_grad()
def compute_csm(raw: torch.Tensor, probe: Probe, K: int, overlap: float, f: Union[torch.Tensor, List[float], float], n_bin: int = 1, n_zp: int = 0) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Calcule la Cross-Spectral Matrix (CSM) par la méthode des snapshots.
    
    Args:
        raw: Tenseur [Nt, Ne] (Temps, Éléments)
        probe.fs: Fréquence d'échantillonnage
        K: Nombre de snapshots souhaités
        overlap: Pourcentage de recouvrement (0.0 à 1.0)
        f: Liste ou tenseur des fréquences de reconstruction
        n_bin: Nombre de bins fréquentiels autour de f
        n_zp: Nombre de points pour la FFT (Zero-padding)
        
    Returns:
        CSM: [F, Ne, Ne] (Moyennée sur les snapshots)
        axf: Fréquences réelles correspondant aux bins sélectionnés
    """
    if raw.dtype != torch.float32:
        raw = raw.to(torch.float32)

    fs=probe.fs
    device = raw.device
    Nt, Ne = raw.shape
    f = torch.as_tensor(f, device=device, dtype=torch.float32)

    # 1. Définition des variables de fenêtrage (Logique MATLAB)
    nw = int(Nt // (K * (1 - overlap) + overlap))
    lag = int(round((1 - overlap) * nw))
    
    # Calcul des indices de départ des snapshots
    starts = torch.arange(0, Nt - nw + 1, lag, device=device)
    n_snapshots = len(starts)

    # 2. Gestion du Zero-Padding (Nzp)
    if n_zp == 0:
        delta_f = 0.01 * f.min()
        n_zp = int(round(fs / delta_f.item()))
        if n_zp < nw:
            n_zp = nw
            
    # On arrondit à la puissance de 2 supérieure pour la vitesse FFT
    n_zp = int(2**np.ceil(np.log2(n_zp)))

    # 3. Identification des bins fréquentiels
    vec_freq = torch.arange(n_zp, device=device,dtype=torch.float32) * (fs / n_zp)
    
    # Trouver les indices les plus proches des fréquences f
    # On utilise broadcasting pour comparer toutes les fréquences d'un coup
    diff = torch.abs(vec_freq.view(1, -1) - f.view(-1, 1))
    idx_center = torch.argmin(diff, dim=1)
    
    # Gestion du multi-bin (n_bin)
    if n_bin > 1:
        offsets = torch.arange(-(n_bin // 2), n_bin // 2 + (n_bin % 2), device=device)
        indices = idx_center.view(-1, 1) + offsets.view(1, -1)
    else:
        indices = idx_center.view(-1, 1)

    # Fréquences réelles des bins
    axf = vec_freq[indices]

    # 4. Préparation de la fenêtre (Tukey/Hann)
    # On utilise une fenêtre de Hann (proche du Tukey MATLAB par défaut)
    win = torch.hann_window(nw, periodic=False, device=device, dtype=torch.float32).view(-1, 1)

    # 5. Calcul des FFT par snapshots
    # On va stocker les spectres complexes : [n_snapshots, n_freq_indices, Ne]
    #tf_f = torch.zeros((n_snapshots, indices.numel(), Ne), dtype=torch.complex64, device=device)
    '''
    for i, start in enumerate(starts):
        # Extraction du snapshot + application de la fenêtre
        snapshot = raw[start : start + nw, :] * win
        
        # FFT sur n_zp points
        full_fft = torch.fft.fft(snapshot, n=n_zp, dim=0)
        
        # On ne garde que les bins qui nous intéressent
        # indices.flatten() contient tous les bins pour toutes les fréquences demandées
        tf_f[i] = full_fft[indices.flatten(), :]'''
    # raw shape: [Nt, Ne]
    snapshots = raw.unfold(dimension=0, size=nw, step=lag) # [n_snapshots, Ne, nw]
    snapshots = snapshots.permute(0, 2, 1) # [n_snapshots, nw, Ne]

    # Application de la fenêtre (broadcast)
    snapshots = snapshots * win.view(1, -1, 1)

    # FFT sur tout le batch d'un coup ! 🔥
    full_fft = torch.fft.fft(snapshots, n=n_zp, dim=1)

    # Extraction des bins
    tf_f = full_fft[:, indices.flatten(), :] # [n_snapshots, n_bins, Ne]

    # 6. Calcul de la CSM : Moyenne de (X * X^H)
    # X est [n_snapshots, n_bins, Ne]
    # On veut CSM : [n_bins, Ne, Ne]
    
    # On réorganise pour le produit extérieur batché
    # [n_bins, n_snapshots, Ne]
    S = tf_f.transpose(0, 1)
    
    # Produit extérieur batché : (bin, snap, Ne, 1) x (bin, snap, 1, Ne)
    # On moyenne sur la dimension 'snap' (n_snapshots)
    csm_per_bin = torch.einsum('bse,bsm->bem', S, S.conj()) / n_snapshots
    
    # Si nBin > 1, on regroupe et on moyenne par fréquence cible
    if n_bin > 1:
        # On redimensionne : [F, nBin, Ne, Ne]
        csm_reshaped = csm_per_bin.view(len(f), n_bin, Ne, Ne)
        # On moyenne sur les bins pour chaque fréquence : [F, Ne, Ne]
        csm_final = csm_reshaped.mean(dim=1)
        # On prend la fréquence centrale pour le steering vector
        axf_final = axf[:, n_bin // 2] 
    else:
        csm_final = csm_per_bin
        axf_final = axf.flatten()

    return csm_final, axf_final

from functools import wraps

def validation_probe(func):
    @wraps(func) 
    def wrapper(self, *args, **kwargs):
        # 1. On exécute l'init (ou la fonction décorée)
        result = func(self, *args, **kwargs)
        
        # 2. On effectue les vérifications après l'assignation
        if self.fs <= 0:
            raise ValueError("The sampling frequency must be defined positive.")
        
        # 3. TRÈS IMPORTANT : On retourne le résultat
        return result
        
    # 4. TRÈS IMPORTANT : On retourne l'objet fonction wrapper
    return wrapper

class Timer:
    """
    Accurate GPU/CPU timing context manager for PyTorch solvers.
    """
    def __init__(self, name="Solver", device="cpu"):
        self.name = name
        self.device = torch.device(device)
        self.elapsed_ms = 0.0

    def __enter__(self):
        if self.device.type == "cuda":
            torch.cuda.synchronize()
        self.start = time.perf_counter()
        return self

    def __exit__(self, *args):
        if self.device.type == "cuda":
            torch.cuda.synchronize()
        self.elapsed_ms = (time.perf_counter() - self.start) * 1000.0
        print(f"[{self.name}] Execution time: {self.elapsed_ms:.2f} ms ({self.elapsed_ms/1000.0:.3f} s)")
